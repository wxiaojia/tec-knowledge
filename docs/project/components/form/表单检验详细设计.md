### 生成1-N的整数的这个正则表达式
```js
function generateRegex(maxValue) {
  const strMax = maxValue.toString();
  const maxLength = strMax.length;

  // 基本的正则构造
  let regex = '^';

  // 针对每一位数进行处理
  for (let i = 1; i <= maxLength; i++) {
    if (i === maxLength) {
      // 对最大数的范围进行限制
      const prefix = strMax.slice(0, i-1);
      const suffix = strMax.slice(i-1);
      regex += `(?:${prefix}[0-${suffix}])$`;
    } else {
      regex += `[1-9]\\d{0,${i-1}}`;
    }
  }

  return regex;
}
```
